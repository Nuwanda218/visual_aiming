# V2 Slim Core Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a minimal V2 runtime skeleton that can process frames through a detector, pipeline, and output backend without carrying V1 GUI, diagnostics, real mouse, or advanced algorithm complexity.

**Architecture:** V2 lives beside V1 as a separate package named `visual_aiming_v2`, with a separate CLI file `main_v2.py`. The first working loop is `FrameSource -> Detector -> Pipeline -> Output`, using fake inputs for tests and a video-file source for reproducible manual runs. V1 remains available as a reference but is not imported by V2 except where a later task explicitly wraps YOLO.

**Tech Stack:** Python 3.12, standard library `unittest`, `dataclasses`, `typing.Protocol`, optional OpenCV for video input, existing `ultralytics` model only through a V2 adapter.

---

This plan implements only the V2 minimal skeleton. It does not implement GUI configuration, interactive video debug windows, full log analytics, real mouse output, sticky target switching, alpha-beta prediction, or advanced control algorithms. from __future__ import annotations

```python
from dataclasses import dataclass
from typing import Any, Optional, Sequence, Tuple

Point = Tuple[int, int]


@dataclass(slots=True)
class Frame:
    sequence: int
    image: Any
    timestamp: float
    crosshair: Point


@dataclass(slots=True)
class Detection:
    x: int
    y: int
    w: int
    h: int
    confidence: float
    label: str = "target"

    @property
    def center(self) -> Point:
        return (self.x + self.w // 2, self.y + self.h // 2)


@dataclass(slots=True)
class Command:
    dx: int = 0
    dy: int = 0
    mode: str = "none"
    reason: str = "noop"

    @classmethod
    def noop(cls, reason: str) -> "Command":
        return cls(dx=0, dy=0, mode="none", reason=reason)


@dataclass(slots=True)
class TickResult:
    frame: Frame
    detections: Sequence[Detection]
    selected: Optional[Detection]
    command: Command
```

```python
from __future__ import annotations

import math
from typing import Sequence

from visual_aiming_v2.ports import Detector, Output
from visual_aiming_v2.schemas import Command, Detection, Frame, TickResult


class Pipeline:
    def __init__(self, detector: Detector, output: Output) -> None:
        self.detector = detector
        self.output = output

    def tick(self, frame: Frame) -> TickResult:
        detections = list(self.detector.detect(frame))
        selected = self._select_nearest(detections, frame.crosshair)
        command = self._command_for(selected, frame)
        result = TickResult(frame=frame, detections=detections, selected=selected, command=command)
        self.output.apply(command, result)
        return result

    def _select_nearest(self, detections: Sequence[Detection], crosshair: tuple[int, int]) -> Detection | None:
        if not detections:
            return None
        cx, cy = crosshair
        return min(detections, key=lambda item: math.hypot(item.center[0] - cx, item.center[1] - cy))

    def _command_for(self, detection: Detection | None, frame: Frame) -> Command:
        if detection is None:
            return Command.noop("no_target")
        dx = detection.center[0] - frame.crosshair[0]
        dy = detection.center[1] - frame.crosshair[1]
        if dx == 0 and dy == 0:
            return Command.noop("deadzone")
        return Command(dx=dx, dy=dy, mode="relative", reason="tracking")
```

```python
from __future__ import annotations

from pathlib import Path
from typing import Iterable, Optional

from visual_aiming_v2.schemas import Frame, Point


class MemoryFrameSource:
    def __init__(self, frames: Iterable[Frame]) -> None:
        self._frames = list(frames)
        self._index = 0
        self.closed = False

    def read(self) -> Optional[Frame]:
        if self._index >= len(self._frames):
            return None
        frame = self._frames[self._index]
        self._index += 1
        return frame

    def close(self) -> None:
        self.closed = True


class VideoFileFrameSource:
    def __init__(self, video_path: str | Path, crosshair: Point = (0, 0)) -> None:
        import cv2

        self.path = Path(video_path)
        self.crosshair = crosshair
        self.capture = cv2.VideoCapture(str(self.path))
        if not self.capture.isOpened():
            raise FileNotFoundError(f"Cannot open video file: {self.path}")
        fps = self.capture.get(cv2.CAP_PROP_FPS)
        self._frame_dt = 1.0 / fps if fps and fps > 0 else 1.0 / 30.0
        self._sequence = 0

    def read(self) -> Optional[Frame]:
        ok, image = self.capture.read()
        if not ok:
            return None
        sequence = self._sequence
        self._sequence += 1
        return Frame(sequence=sequence, image=image, timestamp=sequence * self._frame_dt, crosshair=self.crosshair)

    def close(self) -> None:
        self.capture.release()
```

```python
from __future__ import annotations

from pathlib import Path
from typing import Sequence

from visual_aiming_v2.schemas import Detection, Frame


class StaticDetector:
    def __init__(self, detections: Sequence[Detection]) -> None:
        self._detections = list(detections)

    def detect(self, frame: Frame) -> Sequence[Detection]:
        return list(self._detections)


class YoloDetector:
    def __init__(self, model_path: str | Path, confidence: float = 0.5) -> None:
        self.model_path = str(model_path)
        self.confidence = confidence
        self._model = None

    def detect(self, frame: Frame) -> Sequence[Detection]:
        if self._model is None:
            from ultralytics import YOLO

            self._model = YOLO(self.model_path)
        results = self._model.predict(frame.image, conf=self.confidence, verbose=False)
        detections: list[Detection] = []
        for result in results:
            boxes = getattr(result, "boxes", None)
            if boxes is None:
                continue
            for box in boxes:
                xyxy = box.xyxy[0].tolist()
                x1, y1, x2, y2 = [int(round(value)) for value in xyxy]
                confidence = float(box.conf[0]) if box.conf is not None else 0.0
                detections.append(Detection(x=x1, y=y1, w=max(0, x2 - x1), h=max(0, y2 - y1), confidence=confidence))
        return detections
```

```python
from __future__ import annotations

from visual_aiming_v2.pipeline import Pipeline
from visual_aiming_v2.ports import Detector, FrameSource, Output
from visual_aiming_v2.schemas import TickResult


def run(source: FrameSource, detector: Detector, output: Output, max_frames: int | None = None) -> list[TickResult]:
    pipeline = Pipeline(detector=detector, output=output)
    results: list[TickResult] = []
    try:
        while max_frames is None or len(results) < max_frames:
            frame = source.read()
            if frame is None:
                break
            results.append(pipeline.tick(frame))
        return results
    finally:
        source.close()
        output.close()
```

```python
from __future__ import annotations

import argparse
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parent
SRC = ROOT / "src"
if str(SRC) not in sys.path:
    sys.path.insert(0, str(SRC))


def parse_args(argv=None):
    parser = argparse.ArgumentParser(description="Slim V2 visual aiming runtime")
    parser.add_argument("--video", required=True, help="Video file to process")
    parser.add_argument("--model", default="models/best.pt", help="YOLO model path")
    parser.add_argument("--output", choices=["null", "log"], default="null", help="Safe output backend")
    parser.add_argument("--max-frames", type=int, default=0, help="Maximum frames to process, 0 means all")
    return parser.parse_args(argv)


def main(argv=None) -> int:
    args = parse_args(argv)

    from visual_aiming_v2.app import run
    from visual_aiming_v2.detectors import YoloDetector
    from visual_aiming_v2.frame_sources import VideoFileFrameSource
    from visual_aiming_v2.outputs import LogOutput, NullOutput

    source = VideoFileFrameSource(args.video)
    detector = YoloDetector(args.model)
    output = LogOutput() if args.output == "log" else NullOutput()
    max_frames = None if args.max_frames <= 0 else args.max_frames
    results = run(source=source, detector=detector, output=output, max_frames=max_frames)
    print(f"Processed {len(results)} frames")
    return 0

    raise SystemExit(main())
```
